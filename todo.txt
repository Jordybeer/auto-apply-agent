--- CRITICAL ---

[ ] push_subscriptions has no RLS or schema migration
    Table was created manually without row-level security.
    Any authenticated user can read/delete another user's push subscription.
    Fix: add supabase/migrations/20260415_push_subscriptions.sql
    CREATE TABLE IF NOT EXISTS public.push_subscriptions (
      id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
      user_id uuid NOT NULL UNIQUE REFERENCES auth.users(id) ON DELETE CASCADE,
      subscription jsonb NOT NULL,
      created_at timestamptz DEFAULT now()
    );
    ALTER TABLE public.push_subscriptions ENABLE ROW LEVEL SECURITY;
    CREATE POLICY "users manage own subscription" ON public.push_subscriptions
      FOR ALL USING (auth.uid() = user_id);
